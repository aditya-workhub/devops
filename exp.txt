pipeline:
pipeline {
    agent any
    stages {
        stage('Compile') {
            steps {
                writeFile file: 'Addition.java', text: '''
public class Addition {
    public static void main(String[] args) {
        int a = 10, b = 20;
        int sum = a + b;
        System.out.println("The sum is: " + sum);
    }
}
'''
                bat 'javac Addition.java'
            }
        }
        stage('Run') {
            steps {
                bat 'java Addition'
            }
        }
    }
}

exp 5 :

pipeline:
pipeline {
    agent any

    environment {
        DEPLOY_DIR = 'C:\\xampp\\htdocs\\mywebsite'
    }

    stages {
        stage('Clone Repository') {
            steps {
                git credentialsId: 'github-creds', branch: 'master', url: 'https://github.com/adityasync/exp5_devops'
            }
        }

        stage('Prepare Deployment Directory') {
            steps {
                bat """
                if not exist "%DEPLOY_DIR%" mkdir "%DEPLOY_DIR%"
                """
            }
        }

        stage('Deploy HTML Files to XAMPP') {
            steps {
                bat """
                copy /Y index.html "%DEPLOY_DIR%"
                copy /Y about.html "%DEPLOY_DIR%"
                copy /Y contact.html "%DEPLOY_DIR%"
                """
            }
        }

        stage('Deployment Completed') {
            steps {
                echo "Visit: http://localhost/mywebsite/index.html"
            }
        }
    }

    post {
        success {
            echo '✅ CI/CD completed successfully!'
        }
        failure {
            echo '❌ Deployment failed.'
        }
    }
}
